import pandas as pd import numpy as np import plotly.graph_objects as go # -------------------- pitch_labels = dict(FF='FF', ST='SW', CH='CH', SL='SL', SI='SI') pitch_colors = { 'FF': 'red', 'SW': 'purple', 'CH': 'green', 'SL': 'blue', 'SI': 'brown' } strike_zone_label = "Strike Zone" desired_pitches = list(pitch_labels.keys()) # -------------------- # Load data df = pd.read_csv("statcast_2024_all_pitches.csv") pitcher_id = 693433 df = df[(df['pitcher'] == pitcher_id) & (df['pitch_type'].isin(desired_pitches))] # Flip lateral positions to correct mound side (right side positive) df['release_pos_x'] = -df['release_pos_x'] df['vx0'] = -df['vx0'] df['ax'] = -df['ax'] # Calculate release_pos_y from release_extension df['release_pos_y'] = 60.5 - df['release_extension'] def compute_vaa(row): y0 = row['release_pos_y'] vy0 = row['vy0'] ay = row['ay'] z0 = row['release_pos_z'] vz0 = row['vz0'] az = row['az'] if np.isclose(ay, 0): if vy0 == 0: return np.nan t = -y0 / vy0 if t <= 0: return np.nan else: a = 0.5 * ay b = vy0 c = y0 disc = b**2 - 4*a*c if disc < 0: return np.nan t = (-b - np.sqrt(disc)) / (2*a) if t < 0: t = (-b + np.sqrt(disc)) / (2*a) if t <= 0: return np.nan vy_plate = vy0 + ay * t vz_plate = vz0 + az * t return np.degrees(np.arctan2(vz_plate, -vy_plate)) # flipped vy_plate sign df['vaa'] = df.apply(compute_vaa, axis=1) # Group by pitch_type and average necessary columns avg_df = df.groupby('pitch_type').agg( release_pos_x=('release_pos_x', 'mean'), release_pos_y=('release_pos_y', 'mean'), release_pos_z=('release_pos_z', 'mean'), vx0=('vx0', 'mean'), vy0=('vy0', 'mean'), vz0=('vz0', 'mean'), ax=('ax', 'mean'), ay=('ay', 'mean'), az=('az', 'mean'), vaa=('vaa', 'mean') ).reset_index() fig = go.Figure() for _, row in avg_df.iterrows(): ptype = row['pitch_type'] x0 = row['release_pos_x'] y0 = row['release_pos_y'] z0 = row['release_pos_z'] vx0 = row['vx0'] vy0 = row['vy0'] vz0 = row['vz0'] ax = row['ax'] ay = row['ay'] az = row['az'] vaa = row['vaa'] # Solve time to plate (y=0) a = 0.5 * ay b = vy0 c = y0 disc = b**2 - 4*a*c if disc < 0: continue t = (-b - np.sqrt(disc)) / (2*a) if t < 0: t = (-b + np.sqrt(disc)) / (2*a) if t <= 0: continue steps = 50 t_vals = np.linspace(0, t, steps) y_vals = y0 + vy0 * t_vals + 0.5 * ay * t_vals ** 2 z_vals = z0 + vz0 * t_vals + 0.5 * az * t_vals ** 2 x_vals = x0 + vx0 * t_vals + 0.5 * ax * t_vals ** 2 # lateral position with flipped sign fig.add_trace(go.Scatter3d( x=x_vals, y=y_vals, z=z_vals, mode='lines', line=dict(color=pitch_colors.get(ptype, 'gray'), width=5), name=f"{pitch_labels.get(ptype, ptype)} (VAA: {vaa:.1f}°)", hovertemplate=( f"Pitch: {pitch_labels.get(ptype, ptype)}
" "Distance to Plate: %{y:.2f} ft
" "Height: %{z:.2f} ft
" "Lateral Pos: %{x:.2f} ft
" f"VAA: {vaa:.1f}°" ) )) # Strike zone box outline sz_width = 1.5 sz_height_top = 3.0 sz_height_bottom = 1.5 sz_depth = 0.17 strike_zone_lines = [ ([sz_width/2, -sz_width/2, -sz_width/2, sz_width/2, sz_width/2], [0, 0, 0, 0, 0], [sz_height_bottom, sz_height_bottom, sz_height_bottom, sz_height_bottom, sz_height_bottom]), ([sz_width/2, -sz_width/2, -sz_width/2, sz_width/2, sz_width/2], [sz_depth, sz_depth, sz_depth, sz_depth, sz_depth], [sz_height_top, sz_height_top, sz_height_top, sz_height_top, sz_height_top]), ([sz_width/2, sz_width/2], [0, sz_depth], [sz_height_bottom, sz_height_top]), ([-sz_width/2, -sz_width/2], [0, sz_depth], [sz_height_bottom, sz_height_top]), ([-sz_width/2, sz_width/2], [0, 0], [sz_height_bottom, sz_height_bottom]), ([-sz_width/2, sz_width/2], [sz_depth, sz_depth], [sz_height_top, sz_height_top]) ] for line in strike_zone_lines: fig.add_trace(go.Scatter3d( x=line[0], y=line[1], z=line[2], mode='lines', line=dict(color='black', width=2), showlegend=False )) fig.update_layout( scene=dict( xaxis_title='Lateral Position (ft)', yaxis_title='Distance to Plate (ft)', zaxis_title='Height (ft)', yaxis=dict(autorange='reversed'), aspectratio=dict(x=1, y=3, z=1), ), title=f"3D Pitch Trajectories — Bryan Woo - 2024)", legend=dict(x=1.1, y=0.5), margin=dict(l=0, r=250, b=0, t=50) ) fig.write_html("pitch_3d_trajectory_BW.html", include_plotlyjs=True, full_html=True) print("Saved 3D pitch trajectory plot as 'pitch_3d_trajectory_flipped.html'. Open this file in a web browser to view.") fig.show()
